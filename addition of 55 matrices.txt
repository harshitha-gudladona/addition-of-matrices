/*addition of 5*5 matrices*/
# include<stdio.h>
int main()
{ 
    int A[5][5],B[5][5],C[5][5];
    int i,j;
    
    printf("enter the elements of first matrix:\n");
    for(i=0;i<5;i++)
    for(j=0;j<5;j++)
    scanf("%d",&A[i][j]);
    
    printf("enter the elements of second matrix:\n");
    for(i=0;i<5;i++)
    for(j=0;j<5;j++)
    scanf("%d",&B[i][j]);
    
    for(i=0;i<5;i++)
    for(j=0;j<5;j++)
    C[i][j]=A[i][j]+B[i][j];
    
    printf("sum of two matrices:\n");
    for(i=0;i<5;i++)
    {
        for(j=0;j<5;j++)
        {
        printf("%d\t",C[i][j]);
        }
        printf("\n");
    }
        
    
    
    return 0;
}
